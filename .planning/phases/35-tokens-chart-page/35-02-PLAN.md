**Data transformation — adapt to actual Phase 33 API shape:**

After fetching token data, transform it into chart.js datasets. The general pattern:

```javascript
function resolveSessionLabel(session) {
  if (session.userLabel) return session.userLabel
  if (session.customTitle) return session.customTitle
  return (session.sessionId || '').slice(0, 8) + '...'
}

function buildTokenSeries(messages, mode) {
  // messages = array of per-assistant-message token objects from API
  // Extract the token count field (adapt field name from Phase 33 API)
  if (mode === 'cumulative') {
    let running = 0
    return messages.map(m => {
      running += (m.input_tokens || 0) + (m.output_tokens || 0)
      return running
    })
  }
  // per-message mode
  return messages.map(m => (m.input_tokens || 0) + (m.output_tokens || 0))
}
```

**chartData computed:**

```javascript
const chartData = computed(() => {
  if (!tokensData.value?.sessions?.length) return { labels: [], datasets: [] }

  const sessions = tokensData.value.sessions
  // Determine max message count across all sessions for shared x-axis
  const maxMessages = Math.max(...sessions.map(s => (s.messages || s.tokenMessages || []).length), 0)
  const labels = Array.from({ length: maxMessages }, (_, i) => i)

  // Per-session datasets
  const sessionDatasets = sessions.map((session, idx) => {
    const messages = session.messages || session.tokenMessages || []
    const series = buildTokenSeries(messages, viewMode.value)
    return {
      label: resolveSessionLabel(session),
      data: series.concat(Array(Math.max(0, maxMessages - series.length)).fill(null)),
      borderColor: projectColor(session.projectPath || session._projectPath || ''),
      backgroundColor: 'transparent',
      borderWidth: 2,
      pointRadius: 2,
      tension: 0,
      hidden: hiddenSessions.value.has(idx),
    }
  })

  // Aggregate "All Sessions" line — sum of all VISIBLE sessions at each index
  const aggregateData = Array.from({ length: maxMessages }, (_, i) => {
    let sum = 0
    let hasAny = false
    sessionDatasets.forEach((ds, idx) => {
      if (!hiddenSessions.value.has(idx) && ds.data[i] != null) {
        sum += ds.data[i]
        hasAny = true
      }
    })
    return hasAny ? sum : null
  })

  // Resolve aggregate line color using CSS token for theme support
  const aggregateColor = getComputedStyle(document.documentElement)
    .getPropertyValue('--color-muted').trim() || '#8b949e'

  const aggregateDataset = {
    label: 'All Sessions',
    data: aggregateData,
    borderColor: aggregateColor,
    backgroundColor: 'transparent',
    borderWidth: 3,
    borderDash: [4, 4],
    pointRadius: 0,
    tension: 0,
    hidden: false,
  }

  // Aggregate always first (index 0)
  return {
    labels,
    datasets: [aggregateDataset, ...sessionDatasets],
  }
})
```

**chartOptions computed (depends on isDark for theme reactivity):**

```javascript
function getToken(name) {
  return getComputedStyle(document.documentElement).getPropertyValue(name).trim()
}

const AGGREGATE_DATASET_INDEX = 0

const chartOptions = computed(() => {
  const _ = isDark.value // dependency tracking for theme reactivity
  return {
    responsive: true,
    maintainAspectRatio: false,
    onClick: (event, activeElements) => {
      if (!activeElements.length) return
      const { datasetIndex } = activeElements[0]
      if (datasetIndex === AGGREGATE_DATASET_INDEX) return // ignore aggregate click
      const sessionIdx = datasetIndex - 1 // offset by 1 since aggregate is at index 0
      const session = tokensData.value?.sessions?.[sessionIdx]
      if (session) {
        selectedSession.value = selectedSession.value?.sessionId === session.sessionId ? null : session
      }
    },
    plugins: {
      legend: { display: false }, // use custom HTML legend instead
      tooltip: {
        backgroundColor: getToken('--color-bg-secondary') || '#f6f8fa',
        titleColor: getToken('--color-heading') || '#243846',
        bodyColor: getToken('--color-body-text') || '#3e4d56',
        borderColor: getToken('--color-border') || '#d0d7de',
        borderWidth: 1,
        callbacks: {
          title: (items) => `Message ${items[0]?.label ?? ''}`,
          label: (context) => {
            const value = context.parsed.y
            if (value == null) return null
            return `${context.dataset.label}: ${value.toLocaleString()} tokens`
          },
        },
      },
    },
    scales: {
      x: {
        title: { display: true, text: 'Assistant Message Index', color: getToken('--color-muted') || '#6e7c87' },
        ticks: { color: getToken('--color-muted') || '#6e7c87' },
        grid: { color: getToken('--color-border') || '#d0d7de' },
      },
      y: {
        title: { display: true, text: viewMode.value === 'cumulative' ? 'Cumulative Tokens' : 'Tokens per Message', color: getToken('--color-muted') || '#6e7c87' },
        ticks: {
          color: getToken('--color-muted') || '#6e7c87',
          callback: (value) => {
            if (value >= 1_000_000) return (value / 1_000_000).toFixed(1) + 'M'
            if (value >= 1_000) return (value / 1_000).toFixed(0) + 'K'
            return value
          },
        },
        grid: { color: getToken('--color-border') || '#d0d7de' },
      },
    },
  }
})
```

`toggleSession(idx)` function:
```javascript
function toggleSession(idx) {
  const next = new Set(hiddenSessions.value)
  if (next.has(idx)) next.delete(idx)
  else next.add(idx)
  hiddenSessions.value = next
}
```
