2. Create `src/client/utils/project-colors.js` — extract the COLOR_PALETTE array and projectColor function currently inline in TimelinePage.vue (lines 432-441). Export both as named exports:
   ```javascript
   export const COLOR_PALETTE = [
     '#4e9af1', '#f4a523', '#2ebd6b', '#e05c5c', '#a87fe0',
     '#00c4bc', '#f06292', '#8bc34a', '#ff8f00', '#78909c',
   ]

   export function projectColor(projectPath) {
     let hash = 5381
     for (const char of projectPath) hash = (hash * 33) ^ char.charCodeAt(0)
     return COLOR_PALETTE[Math.abs(hash) % COLOR_PALETTE.length]
   }
   ```
